// Direct Browser Form Submission Handler via Web3Forms API async function handleFormSubmit(e) { e.preventDefault(); const form = document.getElementById('inquiryForm'); const submitBtn = document.getElementById('submitBtn'); const feedback = document.getElementById('formFeedback'); const originalBtnHTML = submitBtn.innerHTML; submitBtn.disabled = true; submitBtn.classList.add('opacity-75', 'cursor-not-allowed'); submitBtn.innerHTML = ` Sending Inquiry Directly... `; feedback.classList.add('hidden'); const formData = new FormData(form); const jsonObject = Object.fromEntries(formData); const jsonString = JSON.stringify(jsonObject); try { const response = await fetch('https://api.web3forms.com/submit', { method: 'POST', headers: { 'Content-Type': 'application/json', 'Accept': 'application/json' }, body: jsonString }); const result = await response.json(); if (response.status === 200 && result.success) { feedback.className = 'p-4 rounded bg-emerald-950/80 border border-emerald-500/50 text-emerald-200 text-xs sm:text-sm text-center block space-y-1'; feedback.innerHTML = 'Inquiry Sent Successfully!Thank you for reaching out. Your project specifications have been submitted directly to our estimating team.'; form.reset(); } else { feedback.className = 'p-4 rounded bg-amber-950/80 border border-amber-500/50 text-amber-200 text-xs sm:text-sm text-center block space-y-1'; feedback.innerHTML = 'Notice: ' + (result.message || 'Submission could not be completed.') + '
Please contact us directly at info@interiorfinishing.net or call (707) 588-8502.'; } } catch (err) { feedback.className = 'p-4 rounded bg-rose-950/80 border border-rose-500/50 text-rose-200 text-xs sm:text-sm text-center block space-y-1'; feedback.innerHTML = 'Transmission Error: Unable to submit from the browser right now. Please email us directly at info@interiorfinishing.net or call (707) 588-8502.'; } finally { submitBtn.disabled = false; submitBtn.classList.remove('opacity-75', 'cursor-not-allowed'); submitBtn.innerHTML = originalBtnHTML; } }
